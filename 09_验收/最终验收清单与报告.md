docker exec dw-etl-mysql mysql -uroot -p<ROOT口令> -t -e "SELECT kpi_code,kpi_value FROM dw.ads_kpi_daily WHERE dt='2026-11-28' ORDER BY kpi_code; SELECT COALESCE(SUM(gmv)/NULLIF(SUM(cost),0),0) AS ROI FROM dw.ads_channel_roi WHERE dt='2026-11-28';"
